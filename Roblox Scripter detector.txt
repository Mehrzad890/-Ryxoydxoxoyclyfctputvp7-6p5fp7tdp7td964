local function makeCorner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = parent
    return c
end

local function makeStroke(parent, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color or CONFIG.COLORS.Border
    s.Thickness = thickness or 1
    s.Transparency = 0.5
    s.Parent = parent
    return s
end

local function tweenObject(obj, props, duration, style, direction)
    return TweenService:Create(obj, TweenInfo.new(
        duration or CONFIG.ANIM_SPEED,
        style or Enum.EasingStyle.Quad,
        direction or Enum.EasingDirection.Out
    ), props)
end

local function createTab(name, icon, order)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.5, -3, 1, -8)
    btn.BackgroundColor3 = CONFIG.COLORS.TabInactive
    btn.BorderSizePixel = 0
    btn.Text = ""
    btn.LayoutOrder = order
    btn.ZIndex = 5
    btn.Parent = TabContainer
    makeCorner(btn, 6)
    local tabIcon = Instance.new("ImageLabel")
    tabIcon.Size = UDim2.new(0, 16, 0, 16)
    tabIcon.Position = UDim2.new(0, 10, 0.5, -8)
    tabIcon.BackgroundTransparency = 1
    tabIcon.Image = icon
    tabIcon.ScaleType = Enum.ScaleType.Fit
    tabIcon.ImageColor3 = CONFIG.COLORS.TextDim
    tabIcon.ZIndex = 6
    tabIcon.Parent = btn
    local tabLabel = Instance.new("TextLabel")
    tabLabel.Size = UDim2.new(1, -34, 1, 0)
    tabLabel.Position = UDim2.new(0, 30, 0, 0)
    tabLabel.BackgroundTransparency = 1
    tabLabel.Text = string.upper(name)
    tabLabel.TextColor3 = CONFIG.COLORS.TextDim
    tabLabel.TextSize = 11
    tabLabel.Font = Enum.Font.GothamBold
    tabLabel.TextXAlignment = Enum.TextXAlignment.Left
    tabLabel.ZIndex = 6
    tabLabel.Parent = btn
    local tab = { button = btn, icon = tabIcon, label = tabLabel, content = nil, name = name }
    tabs[name] = tab
    return tab
end

local function setActiveTab(tabName)
    if activeTab == tabName then return end
    activeTab = tabName
    for name, tab in pairs(tabs) do
        if name == tabName then
            tweenObject(tab.button, { BackgroundColor3 = CONFIG.COLORS.TabActive }):Play()
            tweenObject(tab.icon, { ImageColor3 = CONFIG.COLORS.Accent }):Play()
            tweenObject(tab.label, { TextColor3 = CONFIG.COLORS.Text }):Play()
            if tab.content then tab.content.Visible = true end
        else
            tweenObject(tab.button, { BackgroundColor3 = CONFIG.COLORS.TabInactive }):Play()
            tweenObject(tab.icon, { ImageColor3 = CONFIG.COLORS.TextDim }):Play()
            tweenObject(tab.label, { TextColor3 = CONFIG.COLORS.TextDim }):Play()
            if tab.content then tab.content.Visible = false end
        end
    end
end

local function viewPlayer(playerName)
    local player = Players:FindFirstChild(playerName)
    if not player then 
        if notifications then notifications:show("View Failed", "Player not found.", CONFIG.COLORS.Warning, ICONS.off) end
        return 
    end
    
    if not player.Character then 
        if notifications then notifications:show("View Failed", playerName .. " has no character.", CONFIG.COLORS.Warning, ICONS.off) end
        return 
    end
    
    local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
    if not humanoid then 
        if notifications then notifications:show("View Failed", "Humanoid not found.", CONFIG.COLORS.Warning, ICONS.off) end
        return 
    end
    
    currentViewTarget = playerName
    workspace.CurrentCamera.CameraSubject = humanoid
    UnviewBtn.Visible = true
    if notifications then notifications:show("Viewing Player", "Camera locked to " .. playerName .. ".", CONFIG.COLORS.Accent, ICONS.players) end
end

local function scrollToPlayer(targetName)
    if not detectorEnabled then
        detectorEnabled = true
        ToggleIcon.Image = ICONS.on
        tweenObject(MainFrame, { Size = CONFIG.DEFAULT_SIZE }):Play()
        tweenObject(ShadowFrame, { Size = CONFIG.DEFAULT_SIZE + UDim2.new(0, 12, 0, 12) }):Play()
        TabContainer.Visible = true
    end
    
    PlayersContent.Visible = true
    CreditsContent.Visible = false
    setActiveTab("Players")
    
    local targetCard = playerCards[targetName]
    if not targetCard or not targetCard.card then 
        return 
    end
    
    local startTime = tick()
    while ScrollFrame.AbsoluteCanvasSize.Y <= 0 and tick() - startTime < 1 do
        RunService.Heartbeat:Wait()
    end
    
    local cFrame = targetCard.card
    local sFrame = ScrollFrame
    local canvasOffset = cFrame.AbsolutePosition.Y - sFrame.AbsolutePosition.Y
    local canvasSize = sFrame.AbsoluteCanvasSize.Y
    local viewSize = sFrame.AbsoluteSize.Y
    
    if canvasSize <= 0 or viewSize <= 0 then 
        return 
    end
    
    canvasOffset = canvasOffset - (viewSize / 4)
    canvasOffset = math.clamp(canvasOffset, 0, math.max(0, canvasSize - viewSize))
    
    tweenObject(sFrame, { CanvasPosition = Vector2.new(0, canvasOffset) }):Play()
    tweenObject(targetCard.cardStroke, { Color = CONFIG.COLORS.Accent, Transparency = 0 }):Play()
    task.delay(1.5, function()
        pcall(function() tweenObject(targetCard.cardStroke, { Color = CONFIG.COLORS.Border, Transparency = 0.5 }):Play() end)
    end)
end

            card.BackgroundColor3 = CONFIG.COLORS.CardBackground
            cardStroke.Color = CONFIG.COLORS.Border
            cardStroke.Transparency = 0.5
            playerData[player.Name] = nil
        end
    end)
    playerCards[player.Name] = {
        card = card, cardStroke = cardStroke, statusBar = statusBar,
        statusIcon = statusIcon, statusLbl = statusLbl, userLbl = userLbl,
        dispLbl = dispLbl, studLbl = studLbl, wsLbl = wsLbl,
        flagsLbl = flagsLbl, groundLbl = groundLbl, hoverOverlay = hoverOverlay,
    }
end

local function removeCard(player)
    if playerCards[player.Name] then
        playerCards[player.Name].card:Destroy()
        playerCards[player.Name] = nil
    end
    playerData[player.Name] = nil
    detectedPlayers[player.Name] = nil
    whitelistedPlayers[player.Name] = nil
end

local function isGrounded(hrp, character)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { character }
    params.FilterType = Enum.RaycastFilterType.Exclude
    return workspace:Raycast(hrp.Position, Vector3.new(0, -4.5, 0), params) ~= nil
end

local function checkCollisions(hrp, character)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { character }
    params.FilterType = Enum.RaycastFilterType.Exclude
    local hits = 0
    local dirs = {
        Vector3.new(0, -3, 0),
        Vector3.new(3, 0, 0),
        Vector3.new(-3, 0, 0),
        Vector3.new(0, 0, 3),
        Vector3.new(0, 0, -3)
    }
    for _, dir in ipairs(dirs) do
        if workspace:Raycast(hrp.Position, dir, params) then hits = hits + 1 end
    end
    return hits
end

local function checkServerTeleport()
    local totalPlayers = 0
    local teleportingPlayers = 0
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and playerData[player.Name] then
            local character = player.Character
            if character then
                local hrp = character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    totalPlayers = totalPlayers + 1
                    local data = playerData[player.Name]
                    local now = tick()
                    local dt = now - (data.lastCheck or now)
                    local distance = (hrp.Position - data.lastPosition).Magnitude
                    if dt < 0.5 and distance > CONFIG.TELEPORT_DISTANCE then
                        teleportingPlayers = teleportingPlayers + 1
                    end
                end
            end
        end
    end
    return totalPlayers >= 3 and (teleportingPlayers / totalPlayers) > 0.5
end

local function checkPlayer(player)
    if player == LocalPlayer then return end
    local character = player.Character
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not hrp or not humanoid then return end
    local card = playerCards[player.Name]
    if not card then return end
    if not playerData[player.Name] then
        playerData[player.Name] = { airTime = 0, lastPosition = hrp.Position, lastCheck = tick(), speedSamples = {}, collisionSamples = {}, wasDetected = false }
    end
    local data = playerData[player.Name]
    local vel = hrp.AssemblyLinearVelocity
    local hSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
    local totalSpeed = vel.Magnitude
    local walkSpeed = math.max(humanoid.WalkSpeed, 1)
    local jumpPower = humanoid.JumpPower
    local grounded = isGrounded(hrp, character)
    data.airTime = grounded and 0 or (data.airTime + CONFIG.CHECK_INTERVAL)
    local now = tick()
    local dt = now - (data.lastCheck or now)
    local distance = (hrp.Position - data.lastPosition).Magnitude
    data.lastPosition = hrp.Position
    data.lastCheck = now
    table.insert(data.speedSamples, hSpeed)
    if #data.speedSamples > CONFIG.SAMPLE_COUNT then table.remove(data.speedSamples, 1) end
    local avgSpeed = 0
    for _, s in ipairs(data.speedSamples) do avgSpeed = avgSpeed + s end
    avgSpeed = avgSpeed / #data.speedSamples
    local collisions = checkCollisions(hrp, character)
    table.insert(data.collisionSamples, collisions)
    if #data.collisionSamples > 3 then table.remove(data.collisionSamples, 1) end
    local avgCollision = 0
    for _, c in ipairs(data.collisionSamples) do avgCollision = avgCollision + c end
    avgCollision = avgCollision / #data.collisionSamples
    local teleported = false
    if dt < 0.5 and distance > CONFIG.TELEPORT_DISTANCE and hSpeed < CONFIG.TELEPORT_DISTANCE then
        if not serverTeleporting then
            teleported = true
        end
    end
    card.studLbl.Text = string.format("Speed: %.1f s/s", hSpeed)
    card.wsLbl.Text = string.format("WS: %d", walkSpeed)
    card.groundLbl.Text = grounded and "Grounded" or "Airborne"
    card.groundLbl.TextColor3 = grounded and CONFIG.COLORS.Clean or CONFIG.COLORS.Warning
    if whitelistedPlayers[player.Name] then
        card.statusBar.BackgroundColor3 = CONFIG.COLORS.Accent
        card.statusIcon.Image = ICONS.normal
        card.statusLbl.Text = "WHITELISTED"
        card.statusLbl.TextColor3 = CONFIG.COLORS.Accent
        card.userLbl.TextColor3 = CONFIG.COLORS.Accent
        card.dispLbl.TextColor3 = Color3.fromRGB(130, 160, 255)
        card.flagsLbl.Text = ""
        if data.wasDetected then
            data.wasDetected = false
            detectedPlayers[player.Name] = nil
        end
        return
    end
    local flags = {}
    if avgSpeed > walkSpeed * CONFIG.SPEED_MULTIPLIER and avgSpeed > 16 then table.insert(flags, "Speed") end
    if data.airTime > CONFIG.MAX_AIR_TIME and hSpeed > 5 and vel.Y > -5 and vel.Y < 5 then table.insert(flags, "Fly") end
    if totalSpeed > CONFIG.FLING_THRESHOLD then table.insert(flags, "Fling") end
    if vel.Y > jumpPower * CONFIG.JUMP_MULTIPLIER and vel.Y > 18 then table.insert(flags, "Abnormal Jump") end
    if teleported then table.insert(flags, "Teleport") end
    if avgCollision == 0 and not grounded and hSpeed > 16 and data.airTime > 1.0 and math.abs(vel.Y) < 10 then table.insert(flags, "NoClip") end
    if #flags > 0 then
        card.statusBar.BackgroundColor3 = CONFIG.COLORS.Cheat
        card.statusIcon.Image = ICONS.scripter
        card.statusLbl.Text = "SCRIPTER DETECTED"
        card.statusLbl.TextColor3 = CONFIG.COLORS.Cheat
        card.userLbl.TextColor3 = CONFIG.COLORS.Cheat
        card.dispLbl.TextColor3 = Color3.fromRGB(200, 100, 100)
        card.flagsLbl.Text = table.concat(flags, " | ")
        card.flagsLbl.TextColor3 = CONFIG.COLORS.Cheat
        if not data.wasDetected then
            if not notifCooldowns[player.Name] or (tick() - notifCooldowns[player.Name] > notifCooldown) then
                notifications:show("Cheater Detected!", player.Name .. " flagged: " .. table.concat(flags, ", "), CONFIG.COLORS.Cheat, ICONS.scripter, player.Name)
                notifCooldowns[player.Name] = tick()
            end
            tweenObject(card.hoverOverlay, { BackgroundTransparency = 0.75, BackgroundColor3 = CONFIG.COLORS.Cheat }):Play()
            task.delay(0.6, function()
                pcall(function() tweenObject(card.hoverOverlay, { BackgroundTransparency = 1, BackgroundColor3 = Color3.fromRGB(255, 255, 255) }):Play() end)
            end)
        end
        data.wasDetected = true
        detectedPlayers[player.Name] = true
    else
        card.statusBar.BackgroundColor3 = CONFIG.COLORS.Clean
        card.statusIcon.Image = ICONS.normal
        card.statusLbl.Text = "Normal"
        card.statusLbl.TextColor3 = CONFIG.COLORS.Clean
        card.userLbl.TextColor3 = CONFIG.COLORS.Text
        card.dispLbl.TextColor3 = CONFIG.COLORS.TextDim
        card.flagsLbl.Text = ""
        if data.wasDetected then
            data.wasDetected = false
            detectedPlayers[player.Name] = nil
        end
    end
end

    createCard(player)
    player.CharacterAdded:Connect(function() playerData[player.Name] = nil end)
end

local function handleLocalRespawn(character)
    if not currentViewTarget then return end
    task.wait(0.5)
    local function tryLock()
        local targetPlayer = Players:FindFirstChild(currentViewTarget)
        if targetPlayer and targetPlayer.Character then
            local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                workspace.CurrentCamera.CameraSubject = hum
                UnviewBtn.Visible = true
                return true
            end
        end
        return false
    end
    
    if not tryLock() then
        local start = tick()
        local conn
        conn = RunService.Heartbeat:Connect(function()
            if not currentViewTarget or tick() - start > 5 then
                if conn then conn:Disconnect() end
                return
            end
            if tryLock() then
                if conn then conn:Disconnect() end
            end
        end)
    end
end

if LocalPlayer.Character then
    LocalPlayer.CharacterAdded:Connect(handleLocalRespawn)
else
    LocalPlayer.CharacterAdded:Connect(handleLocalRespawn)
end

    local count = 0
    local detected = 0
    for name in pairs(playerCards) do
        count = count + 1
        if detectedPlayers[name] then detected = detected + 1 end
    end
    PlayerCountLabel.Text = string.format("Players: %d", count)
    DetectedCountLabel.Text = string.format("Detected: %d", detected)
    DetectedCountLabel.TextColor3 = detected > 0 and CONFIG.COLORS.Cheat or CONFIG.COLORS.TextDim
    if not detectorEnabled then return end
    elapsed = elapsed + dt
    if elapsed < CONFIG.CHECK_INTERVAL then return end
    elapsed = 0
    serverTeleporting = checkServerTeleport()
    for _, player in ipairs(Players:GetPlayers()) do pcall(checkPlayer, player) end
end)
